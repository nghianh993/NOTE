package com.example.camunda.connection.interceptor;

import io.grpc.CallOptions;
import io.grpc.Channel;
import io.grpc.ClientCall;
import io.grpc.ClientInterceptor;
import io.grpc.ForwardingClientCall.SimpleForwardingClientCall;
import io.grpc.Metadata;
import io.grpc.MethodDescriptor;

public class AuthHeaderClientInterceptor implements ClientInterceptor {

    private final String clientId;
    private final String clientSecret;
    private final String clientType;

    public AuthHeaderClientInterceptor(String clientId, String clientSecret, String clientType) {
        this.clientId = clientId;
        this.clientSecret = clientSecret;
        this.clientType = clientType;
    }

    @Override
    public <ReqT, RespT> ClientCall<ReqT, RespT> interceptCall(
            MethodDescriptor<ReqT, RespT> method, CallOptions callOptions, Channel next) {
        return new SimpleForwardingClientCall<ReqT, RespT>(next.newCall(method, callOptions)) {
            @Override
            public void start(Listener<RespT> responseListener, Metadata headers) {
                // Inject custom authentication headers into every gRPC request
                headers.put(Metadata.Key.of("client_id", Metadata.ASCII_STRING_MARSHALLER), clientId);
                headers.put(Metadata.Key.of("client_secret", Metadata.ASCII_STRING_MARSHALLER), clientSecret);
                headers.put(Metadata.Key.of("client_type", Metadata.ASCII_STRING_MARSHALLER), clientType);
                super.start(responseListener, headers);
            }
        };
    }
}
