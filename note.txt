 public static CamundaClientProperties load(String resourcePath) {
        Yaml yaml = new Yaml(new Constructor(CamundaClientProperties.class));
        try (InputStream in = CamundaClientProperties.class.getClassLoader().getResourceAsStream(resourcePath)) {
            if (in == null) {
                throw new RuntimeException("Resource not found: " + resourcePath);
            }
            return yaml.load(in);
        } catch (Exception e) {
            throw new RuntimeException("Failed to load configuration from " + resourcePath, e);
        }
    }
