import java.time.Duration;

public class CamundaClientProperties {

    private String mode;
    private Auth auth = new Auth();
    private Zeebe zeebe = new Zeebe();

    // Getters and Setters
    public String getMode() {
        return mode;
    }
    public void setMode(String mode) {
        this.mode = mode;
    }
    public Auth getAuth() {
        return auth;
    }
    public void setAuth(Auth auth) {
        this.auth = auth;
    }
    public Zeebe getZeebe() {
        return zeebe;
    }
    public void setZeebe(Zeebe zeebe) {
        this.zeebe = zeebe;
    }

    public static class Auth {
        private String clientId;
        private String clientSecret;
        private String issuer; // OAuth token endpoint URL

        // Getters and Setters
        public String getClientId() {
            return clientId;
        }
        public void setClientId(String clientId) {
            this.clientId = clientId;
        }
        public String getClientSecret() {
            return clientSecret;
        }
        public void setClientSecret(String clientSecret) {
            this.clientSecret = clientSecret;
        }
        public String getIssuer() {
            return issuer;
        }
        public void setIssuer(String issuer) {
            this.issuer = issuer;
        }
    }

    public static class Zeebe {
        private boolean enabled;
        private String grpcAddress;
        private String restAddress;
        private String audience;
        private Defaults defaults = new Defaults();

        // Getters and Setters
        public boolean isEnabled() {
            return enabled;
        }
        public void setEnabled(boolean enabled) {
            this.enabled = enabled;
        }
        public String getGrpcAddress() {
            return grpcAddress;
        }
        public void setGrpcAddress(String grpcAddress) {
            this.grpcAddress = grpcAddress;
        }
        public String getRestAddress() {
            return restAddress;
        }
        public void setRestAddress(String restAddress) {
            this.restAddress = restAddress;
        }
        public String getAudience() {
            return audience;
        }
        public void setAudience(String audience) {
            this.audience = audience;
        }
        public Defaults getDefaults() {
            return defaults;
        }
        public void setDefaults(Defaults defaults) {
            this.defaults = defaults;
        }

        public static class Defaults {
            private Duration pollInterval;
            private boolean streamEnabled;
            private Duration requestTimeout;
            private int maxJobsActive;
            private Duration timeout;

            // Getters and Setters
            public Duration getPollInterval() {
                return pollInterval;
            }
            public void setPollInterval(Duration pollInterval) {
                this.pollInterval = pollInterval;
            }
            public boolean isStreamEnabled() {
                return streamEnabled;
            }
            public void setStreamEnabled(boolean streamEnabled) {
                this.streamEnabled = streamEnabled;
            }
            public Duration getRequestTimeout() {
                return requestTimeout;
            }
            public void setRequestTimeout(Duration requestTimeout) {
                this.requestTimeout = requestTimeout;
            }
            public int getMaxJobsActive() {
                return maxJobsActive;
            }
            public void setMaxJobsActive(int maxJobsActive) {
                this.maxJobsActive = maxJobsActive;
            }
            public Duration getTimeout() {
                return timeout;
            }
            public void setTimeout(Duration timeout) {
                this.timeout = timeout;
            }
        }
    }
}
