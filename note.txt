CREATE PROCEDURE sp_CalculateRdsCostByInstance
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Các loại instance từ bảng RESULT_RDS_CALCULATE
    WITH InstanceTypesInUse AS (
        SELECT DISTINCT DbInstanceType
        FROM [IDA_CLOUD_COST].[dbo].[RESULT_RDS_CALCULATE]
        WHERE DbInstanceType IS NOT NULL
    ),

    -- 2. Chi phí matched theo instance type (có trong bảng RESULT)
    MatchedCost AS (
        SELECT
            RIGHT([UsageType], LEN([UsageType]) - CHARINDEX(':db.', [UsageType]) - 1) AS DbInstanceType,
            Aws_AccountName,
            SUM([Cost]) AS TotalCost
        FROM [IDA_CLOUD_COST].[dbo].[RDS_COST]
        WHERE [UsageType] LIKE 'APS1-InstanceUsage:db.%'
            AND [UsageType] != 'TOTAL'
            AND RIGHT([UsageType], LEN([UsageType]) - CHARINDEX(':db.', [UsageType]) - 1) IN (
                SELECT DbInstanceType FROM InstanceTypesInUse
            )
        GROUP BY RIGHT([UsageType], LEN([UsageType]) - CHARINDEX(':db.', [UsageType]) - 1), Aws_AccountName
    ),

    -- 3. Tổng uptime theo instance type
    MatchedUptime AS (
        SELECT
            DbInstanceType,
            Aws_AccountName,
            SUM(TotalUptime) AS TotalUptime
        FROM [IDA_CLOUD_COST].[dbo].[RESULT_RDS_CALCULATE]
        WHERE DbInstanceType IS NOT NULL
        GROUP BY DbInstanceType, Aws_AccountName
    ),

    -- 4. Tính đơn giá theo giờ theo account + type
    UnitCost AS (
        SELECT
            m.DbInstanceType,
            m.Aws_AccountName,
            m.TotalCost / u.TotalUptime AS UnitCost
        FROM MatchedCost m
        JOIN MatchedUptime u ON m.DbInstanceType = u.DbInstanceType AND m.Aws_AccountName = u.Aws_AccountName
    ),

    -- 5. Tổng hợp unmatched cost (không có instance type trong RESULT_RDS_CALCULATE)
    UnmatchedCost AS (
        SELECT
            Aws_AccountName,
            SUM(Cost) AS UnmatchedTotalCost
        FROM [IDA_CLOUD_COST].[dbo].[RDS_COST]
        WHERE [UsageType] LIKE 'APS1-InstanceUsage:db.%'
            AND [UsageType] != 'TOTAL'
            AND RIGHT([UsageType], LEN([UsageType]) - CHARINDEX(':db.', [UsageType]) - 1) NOT IN (
                SELECT DbInstanceType FROM InstanceTypesInUse
            )
        GROUP BY Aws_AccountName
    ),

    -- 6. Ghép chi phí cho từng instance
    InstanceCosts AS (
        SELECT
            r.Aws_AccountName,
            r.DbInstanceIdentifier,
            r.DbInstanceType,
            r.TotalUptime,
            u.UnitCost,
            r.TotalUptime * ISNULL(u.UnitCost, 0) AS BaseCost,
            CASE WHEN u.UnitCost IS NULL THEN 1 ELSE 0 END AS IsUnmapped
        FROM [IDA_CLOUD_COST].[dbo].[RESULT_RDS_CALCULATE] r
        LEFT JOIN UnitCost u ON r.DbInstanceType = u.DbInstanceType AND r.Aws_AccountName = u.Aws_AccountName
    ),

    -- 7. Tính số lượng instance hợp lệ mỗi account để chia unmatched cost
    MatchedInstanceCount AS (
        SELECT
            Aws_AccountName,
            COUNT(*) AS InstanceCount
        FROM InstanceCosts
        WHERE IsUnmapped = 0
        GROUP BY Aws_AccountName
    ),

    -- 8. Tổng hợp cuối cùng: cộng base cost + unmatched phân bổ
    FinalCost AS (
        SELECT
            i.Aws_AccountName,
            i.DbInstanceIdentifier,
            i.BaseCost + ISNULL(uc.UnmatchedTotalCost / NULLIF(m.InstanceCount, 0), 0) AS TotalCost
        FROM InstanceCosts i
        LEFT JOIN UnmatchedCost uc ON i.Aws_AccountName = uc.Aws_AccountName
        LEFT JOIN MatchedInstanceCount m ON i.Aws_AccountName = m.Aws_AccountName
    )

    -- 9. Xuất kết quả
    SELECT
        Aws_AccountName,
        DbInstanceIdentifier,
        ROUND(TotalCost, 2) AS FinalCostUSD
    FROM FinalCost
    ORDER BY Aws_AccountName, DbInstanceIdentifier;

END;
