CREATE PROCEDURE [dbo].[sp_CalculateRdsCostByInstance]
AS
BEGIN
    -- Tạo bảng tạm để lưu chi phí đã phân bổ
    IF OBJECT_ID('tempdb..#InstanceCost') IS NOT NULL DROP TABLE #InstanceCost
    IF OBJECT_ID('tempdb..#FinalCost') IS NOT NULL DROP TABLE #FinalCost

    -- 1. Lấy chi phí usage chứa ':db.' (chi phí cụ thể theo instance type)
    SELECT
        rdc.Aws_AccountName,
        REPLACE(SUBSTRING(rdc.UsageType, CHARINDEX(':db.', rdc.UsageType) + 4, LEN(rdc.UsageType)), '-usage', '') AS DbInstanceType,
        SUM(rdc.Cost) AS TotalMatchedCost
    INTO #InstanceCost
    FROM [RDS_COST] rdc
    WHERE rdc.UsageType LIKE '%:db.%'
    GROUP BY rdc.Aws_AccountName,
             REPLACE(SUBSTRING(rdc.UsageType, CHARINDEX(':db.', rdc.UsageType) + 4, LEN(rdc.UsageType)), '-usage', '')

    -- 2. Tính tổng chi phí KHÔNG PHẢI usage ':db.' theo account
    ;WITH UnmatchedCost AS (
        SELECT
            Aws_AccountName,
            SUM(Cost) AS TotalUnmatchedCost
        FROM [RDS_COST]
        WHERE UsageType NOT LIKE '%:db.%'
              AND UsageType <> 'TOTAL'
        GROUP BY Aws_AccountName
    )

    -- 3. Gộp với bảng RESULT_RDS_CALCULATE để tính chi phí theo instance
    SELECT 
        r.Aws_AccountName,
        r.DbInstanceIdentifier,
        r.DbInstanceType,
        r.TotalUptime,
        ISNULL(ic.TotalMatchedCost, 0) AS MatchedCost,
        ISNULL(uc.TotalUnmatchedCost, 0) AS UnmatchedCost,
        COUNT(*) OVER (PARTITION BY r.Aws_AccountName) AS InstanceCountPerAccount
    INTO #FinalCost
    FROM [RESULT_RDS_CALCULATE] r
    LEFT JOIN #InstanceCost ic
        ON r.DbInstanceType = ic.DbInstanceType AND r.Aws_AccountName = ic.Aws_AccountName
    LEFT JOIN (
        SELECT Aws_AccountName, SUM(Cost) AS TotalUnmatchedCost
        FROM [RDS_COST]
        WHERE UsageType NOT LIKE '%:db.%' AND UsageType <> 'TOTAL'
        GROUP BY Aws_AccountName
    ) uc ON r.Aws_AccountName = uc.Aws_AccountName

    -- 4. Tính tổng chi phí cuối cùng
    SELECT
        Aws_AccountName,
        DbInstanceIdentifier,
        CAST((MatchedCost + (UnmatchedCost / NULLIF(InstanceCountPerAccount, 0))) AS DECIMAL(18,2)) AS FinalCostUSD
    FROM #FinalCost
END
